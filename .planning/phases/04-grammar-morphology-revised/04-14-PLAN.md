class _DerivedRowState extends ConsumerState<_DerivedRow> {
  final _meaningCtrl = TextEditingController();

  @override
  Widget build(BuildContext context, WidgetRef ref) {

            ),

          ),

        ),
      ],
    );

  Future<void> _handleMeaningSubmit(String value) async {
    final trimmed = value.trim();
    final dao = ref.read(lexemeDaoProvider);
    if (_promotedLexemeId == null && trimmed.isNotEmpty) {
      // D-57 PROMOTE
      final id = await dao.promoteDerivation(
        parentId: widget.parentLexemeId,
        ruleId: widget.derivation.ruleId,
        gloss: trimmed,
      );
      setState(() => _promotedLexemeId = id);
    } else if (_promotedLexemeId != null && trimmed.isEmpty) {
      // D-57 DEMOTE — confirmation dialog
      final confirmed = await _confirmDemote();
      if (confirmed) {
        await dao.demoteDerivation(lexemeId: _promotedLexemeId!);
        setState(() => _promotedLexemeId = null);
      } else {
        // Restore the previous meaning
        _meaningCtrl.text = /* previous */;
      }
    } else if (_promotedLexemeId != null) {
      // Update an existing promoted row's meaning (D-58: meaning edit does NOT detach)
      await dao.updateLexemeMeaning(_promotedLexemeId!, trimmed);
    }
  }

  Future<void> _handleFormEdit() async {
    if (_promotedLexemeId == null) {
      // Not promoted — editing the form on a computed row is meaningless
      // (the form is derived from the rule). Show a tooltip/snackbar:
      // "Promote this row (add a meaning) before editing the form."
      return;
    }
    // D-58: show implicit-detach warning BEFORE opening the edit dialog
    if (derivedViaRuleId != null) {
      final confirmed = await showDialog<bool>(
        context: context,
        builder: (_) => AlertDialog(
          title: const Text('Unlink from rule?'),
          content: Text(
            'Editing this form will unlink it from rule "${ruleName}". '
            'The row will become standalone — future edits to rule "${ruleName}" will no longer update this form. Continue?',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.of(context).pop(false),
              child: const Text('Cancel'),
            ),
            FilledButton(
              onPressed: () => Navigator.of(context).pop(true),
              child: const Text('Unlink and edit'),
            ),
          ],
        ),
      );
      if (confirmed != true) return;
    }
    // Open a small form-edit dialog (rom + ipa fields) — reuse the existing
    // CellOverrideDialog pattern if applicable, or build a minimal inline editor.
    final edited = await _showFormEditDialog(context);
    if (edited == null) return;
    final dao = ref.read(lexemeDaoProvider);
    await dao.detachFromRule(
      lexemeId: _promotedLexemeId!,
      newIpa: edited.ipa,
      newRom: edited.rom,
    );
  }
}

String? _posAbbrForRule(WidgetRef ref, int ruleId) {
  final rulesAsync = ref.watch(morphologicalRuleListProvider);
  final rule = rulesAsync.asData?.value?.firstWhereOrNull((r) => r.id == ruleId);
  if (rule?.outputPosId == null) return null;
  final posList = ref.watch(posListProvider).asData?.value ?? const [];

class _SuggestionsSection extends ConsumerWidget {
  const _SuggestionsSection({required this.lexeme});
  final Lexeme lexeme;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final theme = Theme.of(context);
    final rulesAsync = ref.watch(morphologicalRuleListProvider);
    final rules = rulesAsync.asData?.value ?? const [];
    final posListAsync = ref.watch(posListProvider);
    final posList = posListAsync.asData?.value ?? const <PartsOfSpeechData>[];
    final pos = posForLexeme(lexeme, posList);
    if (pos == null) return const SizedBox.shrink();

    // Load existing promoted (parent, rule) pairs to hide applied suggestions
    final allLexemesAsync = ref.watch(allLexemeListProvider);
    final allLexemes = allLexemesAsync.asData?.value ?? const [];
    final appliedRuleIds = allLexemes
        .where((lx) =>
            lx.derivedFromLexemeId == lexeme.id && lx.derivedViaRuleId != null)
        .map((lx) => lx.derivedViaRuleId!)
        .toSet();

    // Filter rules: derivational + autoApply=false + inputPos matches + not already applied
    final suggestions = rules.where((r) {
      if (!r.isActive) return false;
      if (r.kind != RuleKind.derivational.dbString) return false;
      if (r.autoApply) return false;  // D-60: only dormant rules become suggestions
      if (r.inputPosId != pos.id) return false;  // D-61 POS match
      if (appliedRuleIds.contains(r.id)) return false;  // hide applied
      return true;
    }).toList();

    if (suggestions.isEmpty) return const SizedBox.shrink();

    return Padding(
      padding: const EdgeInsets.only(top: 12, bottom: 8),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            'Suggestions',
            style: theme.textTheme.bodyMedium?.copyWith(
              fontSize: 13, fontWeight: FontWeight.w600,
            ),
          ),
          const SizedBox(height: 6),
          Wrap(
            spacing: 6,
            runSpacing: 6,
            children: suggestions.map((rule) => ActionChip(
              label: Text(rule.name),
              tooltip: rule.source,
              onPressed: () async {
                final dao = ref.read(lexemeDaoProvider);
                if (dao == null) return;
                // D-60: clicking a chip promotes with empty gloss — user
                // fills in the meaning via the derivation tree row
                await dao.promoteDerivation(
                  parentId: lexeme.id,
                  ruleId: rule.id,
                  gloss: '',
                );
              },
            )).toList(),
          ),
        ],
      ),
    );

class _ParentsSection extends ConsumerWidget {
  const _ParentsSection({required this.lexeme});
  final Lexeme lexeme;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final theme = Theme.of(context);
    final parentsAsync = ref.watch(parentsForLexemeProvider(lexeme.id));
    final parents = parentsAsync.asData?.value ?? const <LexemeParentRow>[];
    if (parents.isEmpty) return const SizedBox.shrink();

    final allLexemesAsync = ref.watch(allLexemeListProvider);
    final allLexemes = allLexemesAsync.asData?.value ?? const [];
    final lexemeById = {for (final lx in allLexemes) lx.id: lx};

    return Padding(
      padding: const EdgeInsets.only(top: 8, bottom: 12),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            'Parents',
            style: theme.textTheme.bodyMedium?.copyWith(
              fontSize: 13, fontWeight: FontWeight.w600,
            ),
          ),
          const SizedBox(height: 6),
          for (final row in parents) _buildParentRow(context, row, lexemeById),
        ],
      ),
    );
  }

  Widget _buildParentRow(
      BuildContext context, LexemeParentRow row, Map<int, Lexeme> lexemeById) {
    final parent = lexemeById[row.parentLexemeId];
    if (parent == null) return const SizedBox.shrink();
    final label = parent.romanization ?? parent.ipa;
    final meaning = parent.meaning ?? '';
    final relationship = row.relationship;
    return Padding(
      padding: const EdgeInsets.only(left: 8, top: 2),
      child: Text(
        relationship != null && relationship.isNotEmpty
            ? '$label ($meaning) — $relationship'
            : '$label ($meaning)',
        style: Theme.of(context).textTheme.bodySmall,
      ),
    );
  }
}
