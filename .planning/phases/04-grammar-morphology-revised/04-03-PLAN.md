**Step 1:** Create `lib/features/grammar/domain/paradigm_cell.dart`:

```dart
import 'inflectional_rule.dart';

/// Result of computing a single paradigm cell via the feature-consumption
/// algorithm (D-10). A sealed hierarchy so callers must handle all three cases.
sealed class ParadigmCell {
  const ParadigmCell();
}

/// The cell was filled successfully. [form] is the final surface form after
/// applying the rule chain in order. [ruleChain] contains each applied rule
/// in application order (most-specific first, then what remains).
class ParadigmFilled extends ParadigmCell {
  const ParadigmFilled({required this.form, required this.ruleChain});
  final String form;
  final List<InflectionalRule> ruleChain;
}

/// No rule (or combination of rules) covers the target feature set.
/// [failureReason] is a human-readable string useful for tooltips
/// (e.g. "No inflectional rule binds {number=PL}" or
///  "Rule '-is' matched but its DSL condition failed on 'kat'").
class ParadigmUncovered extends ParadigmCell {
  const ParadigmUncovered([this.failureReason]);
  final String? failureReason;
}

/// Two or more rules have the same (maximum) specificity and both match the
/// remaining feature set. The user must resolve the conflict (D-12).
class ParadigmAmbiguous extends ParadigmCell {
  const ParadigmAmbiguous(this.tiedRules);
  final List<InflectionalRule> tiedRules;
}
```

import '../../morphology/domain/morphology_dsl.dart';
import '../../morphology/domain/morphology_engine.dart';
import '../../phonology/domain/word_generator.dart'; // for PhonemeInventory type

/// The feature set identifying one paradigm cell.
/// Keys are dimension ids; values are level ids within that dimension.
typedef FeatureSet = Map<int, int>;

/// Compute a single paradigm cell using the feature-consumption algorithm
/// (Phase 4 CONTEXT.md D-10, D-11, D-12, D-13).
///
/// Algorithm (strict most-specific with intra-specificity fall-through — A3):
///
///   1. Pre-filter: drop any rule with `isActive == false` or
///      `bindings.isDerivational` (D-13: unbound never fires).
///   2. While `remaining` feature set is non-empty:
///      a. Collect all candidates whose `bindings.dims` is a non-empty subset
///         of `remaining`. (Non-empty: a rule must consume at least one dim.)
///      b. If candidates empty → return ParadigmUncovered (if no chain yet)
///         or ParadigmFilled(working, chain) (partial chain).
///      c. Sort candidates by specificity descending.
///      d. Take the top specificity group (all candidates at the max specificity).
///      e. If top group has length > 1 AND their binding sets are identical,
///         return ParadigmAmbiguous(top group). Identical binding sets are
///         the only tie that matters — same specificity with different dims
///         means they target different cells and are NOT in conflict here
///         (though tiebreakDetector may still flag them as potentially
///         conflicting on other cells).
///      f. Try to apply each candidate in the top group in order. Apply via
///         parseMorphDsl + engine.applyRule. The FIRST one that returns
///         MorphSuccess wins.
///      g. If ALL top-group candidates return MorphNoMatch, return
///         ParadigmUncovered (strict: do NOT fall through to lower specificity).
///      h. On success, update `working` form, append rule to chain, remove
///         bound dims from remaining, loop.
///   3. When remaining is empty, return ParadigmFilled(working, chain).
ParadigmCell computeParadigmCell({
  required String root,
  required FeatureSet target,
  required List<InflectionalRule> rules,
  required PhonemeInventory inventory,
  MorphologyEngine engine = const MorphologyEngine(),
}) {
  // Pre-filter
  final active = rules.where((r) => r.isActive && r.bindings.isInflectional).toList();

  var working = root;
  var remaining = Map<int, int>.from(target);
  final chain = <InflectionalRule>[];

  while (remaining.isNotEmpty) {
    // Subset match: rule.bindings.dims ⊆ remaining (and non-empty after filter)
    final candidates = active.where((r) => _isSubset(r.bindings.dims, remaining)).toList();
    if (candidates.isEmpty) {
      if (chain.isEmpty) {
        return const ParadigmUncovered('No inflectional rule binds the requested feature set.');
      }
      return ParadigmFilled(form: working, ruleChain: chain);
    }

    // Sort by specificity desc and take the top group.
    candidates.sort((a, b) => b.specificity.compareTo(a.specificity));
    final maxSpec = candidates.first.specificity;
    final topGroup = candidates.where((r) => r.specificity == maxSpec).toList();

    // Ambiguous iff 2+ rules with identical binding sets at the top.
    final identicalBindings = topGroup
        .where((r) => _mapEq(r.bindings.dims, topGroup.first.bindings.dims))
        .toList();
    if (identicalBindings.length > 1 && identicalBindings.length == topGroup.length) {
      return ParadigmAmbiguous(topGroup);
    }
    if (identicalBindings.length > 1) {
      // Partial identical set — still a conflict; report them as tied.
      return ParadigmAmbiguous(identicalBindings);
    }

    // Apply candidates in the top group in list order; first MorphSuccess wins.
    // If all fail → strict: cell uncovered.
    InflectionalRule? winner;
    String? winnerForm;
    String? failureReason;
    for (final c in topGroup) {
      final parsed = parseMorphDsl(c.source, id: c.id, name: c.name);
      if (!parsed.isValid || parsed.rule == null) {
        failureReason ??= 'Rule "${c.name}" has invalid DSL: ${parsed.errorMessage ?? 'parse error'}';
        continue;
      }
      final result = engine.applyRule(parsed.rule!, working, inventory);
      switch (result) {
        case MorphSuccess(:final form):
          winner = c;
          winnerForm = form;
          break;
        case MorphNoMatch(:final reason):
          failureReason ??= 'Rule "${c.name}" did not match: $reason';
      }
      if (winner != null) break;
    }
    if (winner == null) {
      // Strict most-specific: do NOT fall through to lower specificity.
      if (chain.isEmpty) {
        return ParadigmUncovered(failureReason ?? 'All candidate rules failed.');
      }
      return ParadigmFilled(form: working, ruleChain: chain);
    }

    // Apply winner.
    working = winnerForm!;
    chain.add(winner);
    for (final dimId in winner.bindings.dims.keys) {
      remaining.remove(dimId);
    }
  }

  return ParadigmFilled(form: working, ruleChain: chain);
}

bool _isSubset(Map<int, int> sub, Map<int, int> sup) {
  if (sub.isEmpty) return false; // unbound never fires (D-13)
  for (final e in sub.entries) {
    if (sup[e.key] != e.value) return false;
  }
  return true;
}

bool _mapEq(Map<int, int> a, Map<int, int> b) {
  if (a.length != b.length) return false;
  for (final k in a.keys) {
    if (b[k] != a[k]) return false;
  }
  return true;
}

**Step 3:** Create `lib/features/grammar/domain/tiebreak_detector.dart`:

```dart
import 'inflectional_rule.dart';

/// A group of inflectional rules whose binding sets are identical. Any cell
/// that matches one would match all of them → tied. Surfaced live in the
/// rule editor banner (D-12). Used in 04-04.
class TiebreakConflict {
  const TiebreakConflict({required this.bindingKey, required this.rules});
  final String bindingKey; // stable string repr like "10:1,11:1"
  final List<InflectionalRule> rules;
}

/// Finds all groups of 2+ active inflectional rules with identical binding
/// sets (same dim ids AND same level ids for each). Two rules with
/// different POS scopes are still flagged if their dim-bindings are
/// identical — POS filtering happens upstream.
List<TiebreakConflict> findDuplicateSpecificityConflicts(List<InflectionalRule> rules) {
  final byKey = <String, List<InflectionalRule>>{};
  for (final r in rules) {
    if (!r.isActive) continue;
    if (r.bindings.isDerivational) continue;
    final key = _bindingKey(r.bindings.dims);
    byKey.putIfAbsent(key, () => []).add(r);
  }
  return byKey.entries
      .where((e) => e.value.length > 1)
      .map((e) => TiebreakConflict(bindingKey: e.key, rules: e.value))
      .toList();
}

String _bindingKey(Map<int, int> dims) {
  final entries = dims.entries.toList()..sort((a, b) => a.key.compareTo(b.key));
  return entries.map((e) => '${e.key}:${e.value}').join(',');
}
```

import 'dart:convert';

import '../../../db/app_database.dart';

/// Axis configuration for rendering a paradigm table (D-26).
/// rows: the dimension whose levels become table rows (nullable: unused)
/// cols: the dimension whose levels become table columns (nullable: unused)
/// tabs: remaining dimensions become a tab bar / dropdown (empty list = no tabs)
class ParadigmAxes {
  const ParadigmAxes({this.rows, this.cols, this.tabs = const []});

  final int? rows;
  final int? cols;
  final List<int> tabs;

  Map<String, dynamic> toJson() => {
        if (rows != null) 'rows': rows,
        if (cols != null) 'cols': cols,
        'tabs': tabs,
      };

  factory ParadigmAxes.fromJson(Map<String, dynamic> json) {
    final tabsRaw = json['tabs'];
    return ParadigmAxes(
      rows: json['rows'] as int?,
      cols: json['cols'] as int?,
      tabs: tabsRaw is List ? tabsRaw.whereType<int>().toList() : const [],
    );
  }

  factory ParadigmAxes.fromJsonString(String raw) {
    if (raw.isEmpty) return const ParadigmAxes();
    final decoded = jsonDecode(raw);
    if (decoded is! Map) return const ParadigmAxes();
    return ParadigmAxes.fromJson(decoded.cast<String, dynamic>());
  }

  String toJsonString() => jsonEncode(toJson());

  /// Sensible default: first dim = rows, second = cols, rest = tabs.
  factory ParadigmAxes.defaultsFor(List<Dimension> dims) {
    if (dims.isEmpty) return const ParadigmAxes();
    if (dims.length == 1) return ParadigmAxes(rows: dims.first.id);
    return ParadigmAxes(
      rows: dims[0].id,
      cols: dims[1].id,
      tabs: dims.skip(2).map((d) => d.id).toList(),
    );
  }

  @override
  bool operator ==(Object other) =>
      other is ParadigmAxes &&
      other.rows == rows &&
      other.cols == cols &&
      other.tabs.length == tabs.length &&
      List.generate(tabs.length, (i) => tabs[i] == other.tabs[i]).every((e) => e);

  @override
  int get hashCode => Object.hash(rows, cols, Object.hashAll(tabs));
}

/// The computed paradigm for a single lexeme — Cartesian product of the
/// lexeme's POS dimensions (minus any skipped dims) × the feature-consumption
/// engine, one cell per combination.
///
/// Keyed by lexemeId so Riverpod auto-caches per-word.
final computedInflectedParadigmProvider =
    Provider.family<Map<FeatureSet, ParadigmCell>, int>((ref, lexemeId) {
  final lexemeAsync = ref.watch(lexemeByIdProvider(lexemeId));
  final lexeme = lexemeAsync.asData?.value;
  if (lexeme == null) return const {};

  final posListAsync = ref.watch(posListProvider);
  final posList = posListAsync.asData?.value ?? const <PartsOfSpeechData>[];
  final pos = posForLexeme(lexeme, posList);
  if (pos == null) return const {};

  final dimsAsync = ref.watch(dimensionsForPosProvider(pos.id));
  final dims = dimsAsync.asData?.value ?? const <Dimension>[];
  if (dims.isEmpty) return const {};

  // Per-word skipped dimensions (D-07)
  final skipped = _decodeSkippedDims(lexeme.skippedDimensionsJson);
  final active = dims.where((d) => !skipped.contains(d.id)).toList();
  if (active.isEmpty) return const {};

  final rulesAsync = ref.watch(inflectionalRulesForPosProvider(pos.id));
  final rules = rulesAsync.asData?.value ?? const <InflectionalRule>[];

  final inventory = ref.watch(phonemeInventoryProvider);

  final cells = <FeatureSet, ParadigmCell>{};
  for (final featureSet in _cartesian(active)) {
    cells[featureSet] = computeParadigmCell(
      root: lexeme.ipa,
      target: featureSet,
      rules: rules,
      inventory: inventory,
    );
  }
  return cells;
});

/// Cartesian product of dimension levels — one FeatureSet per combination.
Iterable<FeatureSet> _cartesian(List<Dimension> dims) sync* {
  if (dims.isEmpty) {
    yield <int, int>{};
    return;
  }
  final first = dims.first;
  final rest = dims.skip(1).toList();
  final firstLevels = decodeLevelsJson(first.levelsJson);
  for (final level in firstLevels) {
    for (final tail in _cartesian(rest)) {
      yield {first.id: level.id, ...tail};
    }
  }
}

Set<int> _decodeSkippedDims(String? raw) {
  if (raw == null || raw.isEmpty) return const {};
  final decoded = jsonDecode(raw);
  if (decoded is! List) return const {};
  return decoded.whereType<int>().toSet();
}
