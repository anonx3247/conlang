```dart
import 'package:drift/drift.dart';
import '../../../db/app_database.dart';

part 'inflectional_rule_pos_dao.g.dart';

@DriftAccessor(tables: [InflectionalRulePOS])
class InflectionalRulePOSDao extends DatabaseAccessor<AppDatabase>
    with _$InflectionalRulePOSDaoMixin {
  InflectionalRulePOSDao(super.db);

  /// Streams the POS set for a single rule id.
  Stream<Set<int>> watchPosSetForRule(int ruleId) {
    return (select(inflectionalRulePOS)..where((t) => t.ruleId.equals(ruleId)))
        .watch()
        .map((rows) => rows.map((r) => r.posId).toSet());
  }

  /// Streams a map of every rule id -> its POS set. Used by rules_page
  /// grouping (D-56) to render inflectional rules grouped by their POS set.
  Stream<Map<int, Set<int>>> watchAllPosSetsByRuleId() {
    return select(inflectionalRulePOS).watch().map((rows) {
      final map = <int, Set<int>>{};
      for (final row in rows) {
        map.putIfAbsent(row.ruleId, () => <int>{}).add(row.posId);
      }
      return map;
    });
  }

  /// Replace the POS set for a rule atomically. Clears existing junction
  /// rows for the rule, then inserts one row per pos id in [posIds]. An
  /// empty set clears all rows without inserting any.
  Future<void> replaceForRule({required int ruleId, required Set<int> posIds}) async {
    await transaction(() async {
      await (delete(inflectionalRulePOS)..where((t) => t.ruleId.equals(ruleId))).go();
      for (final posId in posIds) {
        await into(inflectionalRulePOS).insert(
          InflectionalRulePOSCompanion.insert(ruleId: ruleId, posId: posId),
          mode: InsertMode.insertOrIgnore,
        );
      }
    });
  }

  /// Delete all junction rows for a rule (used when the rule itself is being deleted).
  /// NOTE: ON DELETE CASCADE on the FK means this is usually not necessary —
  /// deleting the parent MorphologicalRules row already cleans up the junction.
  /// Provided for explicit cleanup scenarios.
  Future<void> deleteAllForRule(int ruleId) {
    return (delete(inflectionalRulePOS)..where((t) => t.ruleId.equals(ruleId))).go();
  }
}
```

```dart
Stream<List<MorphologicalRule>> watchInflectionalRulesForPos(int posId) {
  // v9+: the junction table is authoritative. Legacy input_pos_id is
  // kept as a convenience cache but no longer consulted for inflectional
  // rule lookup (D-55). A single rule can apply to multiple POS via the
  // junction, so a query for POS X returns any rule whose junction row
  // includes posId=X.
  final query = select(morphologicalRules).join([
    innerJoin(
      inflectionalRulePOS,
      inflectionalRulePOS.ruleId.equalsExp(morphologicalRules.id),
    ),
  ])..where(
      morphologicalRules.kind.equals('inflectional') &
          inflectionalRulePOS.posId.equals(posId),
    );
  return query.watch().map((rows) {
    final seen = <int>{};
    final result = <MorphologicalRule>[];
    for (final row in rows) {
      final rule = row.readTable(morphologicalRules);
      if (seen.add(rule.id)) result.add(rule);
    }
    return result;
  });
}
```

The `seen` set handles the case where a rule's junction has multiple rows — the query returns one joined row per junction entry, so deduplication is required.

**Step 2 — In rules_page.dart**, add a grouping helper function (kept private to the file) that takes `(List<MorphologicalRule>, Map<int, Set<int>>, List<PartsOfSpeechData>)` and returns an ordered `List<({String header, List<MorphologicalRule> rules})>`:

```dart
/// D-56 grouping — single-POS groups first (alphabetic by POS name),
/// then multi-POS groups (each set forms one group named after the set,
/// alphabetized within the tier), then an 'Unattached' group for rules
/// with no junction rows.
List<({String header, List<MorphologicalRule> rules})> _groupInflectionalRulesByPosSet({
  required List<MorphologicalRule> rules,
  required Map<int, Set<int>> posSetByRuleId,
  required List<PartsOfSpeechData> posList,
}) {
  final posNameById = {for (final p in posList) p.id: p.name};
  final abbrById = {for (final p in posList) p.id: p.abbreviation};

  String headerFor(Set<int> posIds) {
    if (posIds.isEmpty) return 'Unattached';
    final names = posIds
        .map((id) => posNameById[id] ?? 'POS#$id')
        .toList()
      ..sort();
    return names.join(' + ');
  }

  // Group rules by a canonical frozen-set key
  final byKey = <String, ({Set<int> posIds, List<MorphologicalRule> rules})>{};
  for (final rule in rules) {
    final posIds = posSetByRuleId[rule.id] ?? const <int>{};
    final key = (posIds.toList()..sort()).join(',');
    byKey.putIfAbsent(key, () => (posIds: posIds, rules: <MorphologicalRule>[]))
        .rules.add(rule);
  }

  // Split into single-POS, multi-POS, unattached tiers
  final singleTier = <({String header, List<MorphologicalRule> rules})>[];
  final multiTier = <({String header, List<MorphologicalRule> rules})>[];
  final unattachedTier = <({String header, List<MorphologicalRule> rules})>[];
  for (final group in byKey.values) {
    final header = headerFor(group.posIds);
    final entry = (header: header, rules: group.rules);
    if (group.posIds.isEmpty) {
      unattachedTier.add(entry);
    } else if (group.posIds.length == 1) {
      singleTier.add(entry);
    } else {
      multiTier.add(entry);
    }
  }

  singleTier.sort((a, b) => a.header.compareTo(b.header));
  multiTier.sort((a, b) => a.header.compareTo(b.header));

  return [...singleTier, ...multiTier, ...unattachedTier];
}
```
