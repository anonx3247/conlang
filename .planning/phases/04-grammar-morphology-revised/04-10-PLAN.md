**Step 1 — Create `lib/features/grammar/domain/marker.dart`:**

```dart
import 'feature_bindings.dart';

/// An unmarked-cell declaration for an inflectional paradigm (D-43 / D-44).
///
/// A marker asserts that a feature-binding set produces NO form change at
/// the cell(s) it matches — the root IS the form. Parallel in shape to
/// [InflectionalRule] but:
///  - No DSL / no form output
///  - Always inflectional-scoped (no kind discriminator)
///  - Resolution: D-45 order is override -> rule -> marker -> uncovered
///  - Specificity: D-10 algorithm (dims.length). On exact ties with a rule,
///    the rule wins (D-46) — rules carry an explicit form, markers assert
///    absence.
class MarkerDecl {
  const MarkerDecl({
    required this.id,
    required this.posId,
    required this.bindings,
  });

  final int id;
  final int posId;
  final FeatureBindings bindings;

  int get specificity => bindings.specificity;

  /// True iff this marker's feature bindings are a subset of [target].
  bool matches(Map<int, int> target) {
    if (bindings.dims.isEmpty) return false;  // unbound markers never fire
    for (final entry in bindings.dims.entries) {
      if (target[entry.key] != entry.value) return false;
    }
    return true;
  }
}
```

import '../../../db/app_database.dart';

@DriftAccessor(tables: [Markers])
class MarkerDao extends DatabaseAccessor<AppDatabase> with _$MarkerDaoMixin {
  MarkerDao(super.db);

  Stream<List<MarkerDecl>> watchMarkersForPos(int posId) {
    return (select(markers)..where((t) => t.posId.equals(posId)))
        .watch()
        .map((rows) => rows
            .map((r) => MarkerDecl(
                  id: r.id,
                  posId: r.posId,
                  bindings: r.featureBindings,
                ))
            .toList());
  }

  Future<int> insertMarker({required int posId, required FeatureBindings bindings}) {
    return into(markers).insert(
      MarkersCompanion.insert(
        posId: posId,
        featureBindings: Value(bindings),
      ),
    );
  }

  Future<void> updateMarker(int id, FeatureBindings bindings) {
    return (update(markers)..where((t) => t.id.equals(id)))
        .write(MarkersCompanion(featureBindings: Value(bindings)));
  }

  Future<void> deleteMarker(int id) {
    return (delete(markers)..where((t) => t.id.equals(id))).go();
  }
}

**Step 3 — Extend `computeParadigmCell`** in `paradigm_engine.dart` with the `markers` parameter and the D-45 resolution step. The resolution order per D-45 is:

1. Per-word override (NOT handled in engine — handled at widget layer)
2. Inflectional rules (existing loop — unchanged, but the loop now terminates differently when it exhausts without consuming all dims)
3. **NEW:** If the rule loop ends with `chain.isEmpty` AND markers list is non-empty, search for a matching marker:
   - Filter markers to those whose bindings.dims are a subset of the ORIGINAL target (not the post-consumption `remaining`)
   - Sort by specificity desc
   - Return `ParadigmUnmarked(root: root, source: bestMarker)` — the root is the pre-inflection form (unmarked cells show the bare root)
4. If no marker matches, fall through to `ParadigmUncovered` (existing behavior)

**Critical D-46 tie-breaking detail:** When BOTH a rule AND a marker match at the SAME specificity on the SAME dimensions, the rule wins. The existing algorithm already picks the rule first (step 2), so this is automatic — markers are only consulted when the rule loop produced NO chain. The interesting case is when the rule chain PARTIALLY consumes dims and a marker could fill the remainder: DEFAULT behavior documented in Test 4 = the rule wins and the marker is not consulted for the remainder. This simplification is intentional; future plans can extend it.

```dart
ParadigmCell computeParadigmCell({
  required String root,
  required FeatureSet target,
  required List<InflectionalRule> rules,
  required PhonemeInventory inventory,
  List<RewriteRule> rewriteRules = const [],
  List<MarkerDecl> markers = const [],  // NEW
  MorphologyEngine engine = const MorphologyEngine(),
}) {
  // ... existing rule-consumption loop ...
  // Existing: when loop ends, if chain is non-empty return ParadigmFilled
  //           (with rewrite rules applied). If chain is empty, currently
  //           returns ParadigmUncovered.

  // D-45 STEP 3: before returning ParadigmUncovered, consult markers.
  // Only when the rule loop produced NO chain — rules beat markers per D-46.
  if (chain.isEmpty) {
    final matchingMarkers = markers
        .where((m) => m.matches(target))

/// D-47 render: bare root in muted gray with a trailing ∅ badge.
/// Distinct from ParadigmFilled (normal) and ParadigmUncovered (em-dash).
class _UnmarkedCell extends ConsumerWidget {
  const _UnmarkedCell({required this.root, required this.lexemeId});

  final String root;
  final int lexemeId;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final theme = Theme.of(context);
    final cs = theme.colorScheme;

    final romanize = ref.watch(romanizeProvider);
    final validate = ref.watch(phonotacticValidatorProvider);
    final lexemeAsync = ref.watch(lexemeByIdProvider(lexemeId));
    final lexeme = lexemeAsync.asData?.value;
    final isException = lexeme?.isPhonologicalException ?? false;

    final romText = romanize(root);
    final showRom = romText.isNotEmpty && romText != root;
    final ValidationResult result = isException
        ? const ValidationResult(violations: [])
        : validate(word: root);

    // Build the muted-gray label (rom or ipa depending on romanization).
    Widget label;
    if (showRom) {
      label = Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          // Muted gray treatment — use onSurface with alpha 0.45
          DefaultTextStyle(
            style: theme.textTheme.bodyMedium!.copyWith(
              color: cs.onSurface.withValues(alpha: 0.45),
            ),
            child: ViolationText(text: romText, violations: result.violations),
          ),
          const SizedBox(height: 2),
          Text(
            '[$root]',
            style: theme.textTheme.bodySmall?.copyWith(
              color: cs.onSurface.withValues(alpha: 0.35),
            ),
          ),
        ],
      );
    } else {
      label = DefaultTextStyle(
        style: theme.textTheme.bodyMedium!.copyWith(
          color: cs.onSurface.withValues(alpha: 0.45),
        ),
        child: ViolationText(text: '[$root]', violations: result.violations),
      );
    }

    return Stack(
      children: [
        Padding(
          padding: const EdgeInsets.symmetric(vertical: 4, horizontal: 2),
          child: Center(child: label),
        ),
        // D-47 trailing ∅ badge top-right
        Positioned(
          top: 2,
          right: 2,
          child: Text(
            '∅',
            style: theme.textTheme.labelSmall?.copyWith(
              color: cs.onSurface.withValues(alpha: 0.55),
              fontWeight: FontWeight.w600,
            ),
          ),
        ),
      ],
    );
  }
}
```
