```dart
@DriftAccessor(tables: [MorphologicalRules, MorphologicalRuleExceptions])
class MorphologyDao extends DatabaseAccessor<AppDatabase> with _$MorphologyDaoMixin {
  MorphologyDao(super.db);

  // Rules CRUD
  Stream<List<MorphologicalRule>> watchAllRules() =>
      (select(morphologicalRules)..orderBy([(t) => OrderingTerm.asc(t.ordering)])).watch();

  Future<int> insertRule(MorphologicalRulesCompanion c) =>
      into(morphologicalRules).insert(c);

  Future<bool> updateRule(MorphologicalRulesData r) =>
      update(morphologicalRules).replace(r);

  Future<int> deleteRule(int id) =>
      (delete(morphologicalRules)..where((t) => t.id.equals(id))).go();

  // Exceptions CRUD
  Stream<List<MorphologicalRuleException>> watchExceptionsForRule(int ruleId) =>
      (select(morphologicalRuleExceptions)..where((t) => t.ruleId.equals(ruleId))).watch();

  Stream<List<MorphologicalRuleException>> watchExceptionsForLexeme(int lexemeId) =>
      (select(morphologicalRuleExceptions)..where((t) => t.lexemeId.equals(lexemeId))).watch();

  Future<int> insertException(MorphologicalRuleExceptionsCompanion c) =>
      into(morphologicalRuleExceptions).insert(c);

  Future<int> deleteException(int id) =>
      (delete(morphologicalRuleExceptions)..where((t) => t.id.equals(id))).go();

  // Stale exception detection: exceptions where ruleSourceSnapshot != current rule source
  Future<List<MorphologicalRuleExceptionsData>> findStaleExceptions(int ruleId, String currentSource) async {
    final all = await (select(morphologicalRuleExceptions)..where((t) => t.ruleId.equals(ruleId))).get();
    return all.where((e) => e.ruleSourceSnapshot != currentSource).toList();
  }
}
```

Note: The generated data class names will be `MorphologicalRulesData` and `MorphologicalRuleExceptionsData` (Drift convention: table class name + "Data" suffix). Use these names in the DAO method signatures -- adjust from the above if Drift generates differently.

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../../project/data/project_providers.dart';
import 'morphology_dao.dart';

/// Provides the MorphologyDao for the current project database.
/// Returns null when no project is open.
final morphologyDaoProvider = Provider<MorphologyDao?>((ref) {
  final db = ref.watch(currentDatabaseProvider);
  if (db == null) return null;
  return db.morphologyDao;
});

/// Streams all morphological rules for the current project, ordered by `ordering`.
final morphologicalRuleListProvider = StreamProvider<List<MorphologicalRulesData>>((ref) {
  final dao = ref.watch(morphologyDaoProvider);
  if (dao == null) return Stream.value([]);
  return dao.watchAllRules();
});

/// Streams exceptions for a specific rule.
final morphRuleExceptionsProvider = StreamProvider.family<List<MorphologicalRuleExceptionsData>, int>((ref, ruleId) {
  final dao = ref.watch(morphologyDaoProvider);
  if (dao == null) return Stream.value([]);
  return dao.watchExceptionsForRule(ruleId);
});
```
