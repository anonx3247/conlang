   ```dart
   import 'package:drift/drift.dart';
   import '../../../db/app_database.dart';
   part 'scratchpad_dao.g.dart';

   @DriftAccessor(tables: [ScratchpadPhrases])
   class ScratchpadDao extends DatabaseAccessor<AppDatabase>
       with _$ScratchpadDaoMixin {
     ScratchpadDao(super.db);

     // -----------------------------------------------------------------------
     // Reads
     // -----------------------------------------------------------------------

     /// Watches all phrases ordered newest-first.
     Stream<List<ScratchpadPhrase>> watchAllPhrases() {
       return (select(scratchpadPhrases)
             ..orderBy([(t) => OrderingTerm.desc(t.updatedAt)]))
           .watch();
     }

     /// Reads a single phrase by id.
     Future<ScratchpadPhrase?> getPhraseById(int id) {
       return (select(scratchpadPhrases)..where((t) => t.id.equals(id)))
           .getSingleOrNull();
     }

     // -----------------------------------------------------------------------
     // Writes
     // -----------------------------------------------------------------------

     /// Creates a new phrase with empty text and returns the auto-generated id.
     Future<int> createPhrase() {
       final now = DateTime.now();
       return into(scratchpadPhrases).insert(ScratchpadPhrasesCompanion.insert(
         originalText: const Value(''),
         createdAt: now,
         updatedAt: now,
       ));
     }

     /// Updates the original text of an existing phrase.
     Future<void> updateOriginalText(int id, String text) {
       return (update(scratchpadPhrases)..where((t) => t.id.equals(id))).write(
         ScratchpadPhrasesCompanion(
           originalText: Value(text),
           updatedAt: Value(DateTime.now()),
         ),
       );
     }

     /// Updates the free translation of an existing phrase.
     Future<void> updateFreeTranslation(int id, String? translation) {
       return (update(scratchpadPhrases)..where((t) => t.id.equals(id))).write(
         ScratchpadPhrasesCompanion(
           freeTranslation: Value(translation),
           updatedAt: Value(DateTime.now()),
         ),
       );
     }

     /// Deletes a phrase by id.
     Future<void> deletePhrase(int id) {
       return (delete(scratchpadPhrases)..where((t) => t.id.equals(id))).go();
     }
   }
   ```

1. **Create `lib/features/scratchpad/domain/gloss_match.dart`** — the sealed class hierarchy for token resolution results:
   ```dart
   import '../../grammar/domain/inflectional_rule.dart';
   import '../../../db/app_database.dart';

   /// Result of resolving a single token against the project's lexicon.
   ///
   /// The [glossLabel] getter produces the Leipzig-format label for the
   /// gloss tier of the interlinear display.
   sealed class GlossMatch {
     const GlossMatch();

     /// Human-readable gloss label for the interlinear tier.
     String get glossLabel;

     /// The IPA (phonemic) form for this match. Used for phonotactic
     /// validation and IPA transcription.
     String get ipa;
   }

   /// Token matched an exact root/base lexeme form.
   class ExactLexemeMatch extends GlossMatch {
     const ExactLexemeMatch({required this.lexeme});
     final Lexeme lexeme;

     @override
     String get glossLabel {
       // Closed-class particles use their glossTag (e.g. NEG, DEF, PROG).
       if (lexeme.isClosedClass && lexeme.glossTag != null && lexeme.glossTag!.isNotEmpty) {
         return lexeme.glossTag!;
       }
       // Open-class words use their meaning (e.g. "stone", "eat").
       return lexeme.meaning ?? lexeme.ipa;
     }

     @override
     String get ipa => lexeme.ipa;
   }

   /// Token matched a known inflected paradigm form.
   class InflectedFormMatch extends GlossMatch {
     const InflectedFormMatch({
       required this.lexeme,
       required this.ruleChain,
       required this.levelAbbrs,
     });
     final Lexeme lexeme;
     final List<InflectionalRule> ruleChain;
     /// Leipzig abbreviations for the inflectional feature set,
     /// e.g. ['PST', '3', 'SG']. Built from DimensionLevel.abbr values.
     final List<String> levelAbbrs;

     @override
     String get glossLabel {
       final base = lexeme.meaning ?? lexeme.ipa;
       if (levelAbbrs.isEmpty) return base;
       return '$base.${levelAbbrs.join('.')}';
     }

     @override
     String get ipa => lexeme.ipa; // root IPA — inflected form IPA handled separately
   }

   /// Token matched a derived form computed by the morphology engine.
   class DerivedFormMatch extends GlossMatch {
     const DerivedFormMatch({
       required this.rootLexeme,
       required this.ruleName,
       required this.derivedIpa,
     });
     final Lexeme rootLexeme;
     final String ruleName;
     final String derivedIpa;

     @override
     String get glossLabel {
       final base = rootLexeme.meaning ?? rootLexeme.ipa;
       return '$base.$ruleName';
     }

     @override
     String get ipa => derivedIpa;
   }

   /// Token did not match any known form.
   class UnknownToken extends GlossMatch {
     const UnknownToken({required this.text});
     final String text;

     @override
     String get glossLabel => '?';

     @override
     String get ipa => text;
   }
   ```

2. **Create `lib/features/scratchpad/domain/phrase_tokenizer.dart`** — whitespace tokenizer with resolution:
   ```dart
   import 'gloss_match.dart';

   /// A single token from a phrase with its resolved gloss match.
   class PhraseToken {
     const PhraseToken({
       required this.text,
       required this.match,
       required this.hasViolation,
     });

     /// The original text as typed by the user.
     final String text;

     /// The resolved gloss match (exact, inflected, derived, or unknown).
     final GlossMatch match;

     /// Whether this token violates phonotactic rules.
     final bool hasViolation;

     /// True if the token was not found in the lexicon.
     bool get isUnknown => match is UnknownToken;
   }

   /// Whitespace-splits [input] into tokens and resolves each against
   /// [glossIndex]. Checks phonotactic validity via [validateWord].
   ///
   /// The lookup priority is: exact match > inflected > derived > unknown
   /// (priority is baked into the index at build time — the first match wins).
   ///
   /// Returns an empty list for empty/whitespace-only input.
   List<PhraseToken> tokenizePhrase({
     required String input,
     required Map<String, GlossMatch> glossIndex,
     required bool Function({required String word}) validateWord,
   }) {
     if (input.trim().isEmpty) return const [];

     final words = input.split(RegExp(r'\s+'));
     return words.map((word) {
       final match = glossIndex[word] ?? UnknownToken(text: word);
       // Phonotactic check uses the IPA form from the match.
       // For unknown tokens, use the raw text as-is (it may or may not be IPA).
       final isValid = validateWord(word: match.ipa);
       return PhraseToken(
         text: word,
         match: match,
         hasViolation: !isValid && match is! UnknownToken,
       );
     }).toList();
   }
   ```
   Note: `validateWord` returns `true` when valid. `hasViolation` is only set for known tokens (unknown tokens get amber highlight instead, not red underline — per CONTEXT.md error highlighting decisions).

3. **Create `lib/features/scratchpad/data/gloss_index_provider.dart`** — the pre-computed index. This is the critical architecture piece per RESEARCH.md Pattern 1:
   ```dart
   import 'package:flutter_riverpod/flutter_riverpod.dart';

   import '../../grammar/data/grammar_providers.dart';
   import '../../grammar/data/typology_providers.dart';
   import '../../grammar/domain/dimension_level.dart';
   import '../../grammar/domain/paradigm_cell.dart';
   import '../../lexicon/data/lexeme_providers.dart';
   import '../../phonology/data/romanization_providers.dart';
   import '../../../db/app_database.dart';
   import '../domain/gloss_match.dart';

   /// Pre-computed index mapping every known surface form (IPA and romanization)
   /// to its [GlossMatch] decomposition. The scratchpad parser does O(1) lookups
   /// against this map instead of brute-force paradigm computation per token.
   ///
   /// Rebuilds reactively when the lexicon, grammar rules, or phonology changes.
   ///
   /// Lookup keys include both IPA and romanization for each form, so users can
   /// type in either notation (per Pitfall 4 from RESEARCH.md).
   final glossIndexProvider = Provider<Map<String, GlossMatch>>((ref) {
     final allLexemes = ref.watch(allLexemeListProvider).asData?.value ?? [];
     if (allLexemes.isEmpty) return const {};

     final romanize = ref.watch(romanizeProvider);
     final posListAsync = ref.watch(posListProvider);
     final posList = posListAsync.asData?.value ?? const <PartsOfSpeechData>[];

     final index = <String, GlossMatch>{};

     for (final lex in allLexemes) {
       // Step 1: exact root/base form
       final exactMatch = ExactLexemeMatch(lexeme: lex);
       index.putIfAbsent(lex.ipa, () => exactMatch);
       if (lex.romanization != null && lex.romanization!.isNotEmpty) {
         index.putIfAbsent(lex.romanization!, () => exactMatch);
       }

       // Step 2: inflected forms via paradigm engine
       final paradigm = ref.watch(computedInflectedParadigmProvider(lex.id));
       if (paradigm.isNotEmpty) {
         // Resolve POS to get dimensions for Leipzig labels
         final pos = posForLexeme(lex, posList);
         final dims = pos != null
             ? (ref.watch(dimensionsForPosProvider(pos.id)).asData?.value ??
                 const <Dimension>[])
             : const <Dimension>[];

         for (final entry in paradigm.entries) {
           if (entry.cell case ParadigmFilled(:final form, :final phonemic, :final ruleChain)) {
             // Build Leipzig abbreviation list from featureSet + dimensions
             final levelAbbrs = _buildLevelAbbrs(entry.featureSet, dims);

             final inflMatch = InflectedFormMatch(
               lexeme: lex,
               ruleChain: ruleChain,
               levelAbbrs: levelAbbrs,
             );

             // Key on phonetic surface form (post-rewrite)
             index.putIfAbsent(form, () => inflMatch);
             // Key on romanized form of the phonemic (pre-rewrite) output
             final rom = romanize(phonemic);
             if (rom != phonemic) {
               index.putIfAbsent(rom, () => inflMatch);
             }
             // Also key on the raw phonemic form
             if (phonemic != form) {
               index.putIfAbsent(phonemic, () => inflMatch);
             }
           }
         }
       }

       // Step 3: derived forms via morphology engine
       final derived = ref.watch(computedDerivedFormsProvider(lex.id));
       for (final d in derived) {
         final derMatch = DerivedFormMatch(
           rootLexeme: lex,
           ruleName: d.ruleName,
           derivedIpa: d.derivedIpa,
         );
         index.putIfAbsent(d.derivedIpa, () => derMatch);
         // Also add romanized key for derived forms
         final rom = romanize(d.derivedIpa);
         if (rom != d.derivedIpa) {
           index.putIfAbsent(rom, () => derMatch);
         }
       }
     }

     return index;
   });

   /// Builds a list of Leipzig abbreviations from a feature set by resolving
   /// each dimension ID → level ID → DimensionLevel.abbr.
   List<String> _buildLevelAbbrs(
     FeatureSet featureSet,
     List<Dimension> dims,
   ) {
     final abbrs = <String>[];
     for (final entry in featureSet.entries) {
       final dimId = entry.key;
       final levelId = entry.value;
       // Find the dimension
       final dim = dims.where((d) => d.id == dimId).firstOrNull;
       if (dim == null) continue;
       // Decode levels JSON to find the level's abbreviation
       final levels = decodeLevelsJson(dim.levelsJson);
       final level = levels.where((l) => l.id == levelId).firstOrNull;
       if (level != null) {
         abbrs.add(level.abbr.toUpperCase());
       }
     }
     return abbrs;
   }
   ```
