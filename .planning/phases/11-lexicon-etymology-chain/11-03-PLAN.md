    ```dart
    Widget _buildAddParentSection(BuildContext context, WidgetRef ref) {
      if (!_addingParent) {
        return TextButton.icon(
          onPressed: () => setState(() => _addingParent = true),
          icon: const Icon(Icons.add, size: 16),
          label: const Text('Add parent', style: TextStyle(fontSize: 12)),
        );
      }

      // Read current manual parent IDs
      final manualParentRows = ref.watch(parentsForLexemeProvider(widget.lexeme.id)).asData?.value ?? const [];
      final manualParentIds = manualParentRows.map((r) => r.parentLexemeId).toSet();

      return Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          SearchableParentSelector(
            selectedIds: manualParentIds,
            onChanged: (newIds) {
              final dao = ref.read(lexemeParentsDaoProvider);
              if (dao == null) return;
              // Add new parents
              for (final id in newIds.difference(manualParentIds)) {
                dao.insertParent(childLexemeId: widget.lexeme.id, parentLexemeId: id);
              }
              // Remove deleted parents
              for (final id in manualParentIds.difference(newIds)) {
                dao.deleteParent(childLexemeId: widget.lexeme.id, parentLexemeId: id);
              }
            },
            excludeId: widget.lexeme.id,
          ),
          TextButton(
            onPressed: () => setState(() => _addingParent = false),
            child: const Text('Done', style: TextStyle(fontSize: 12)),
          ),
        ],
      );
    }
    ```
