    ```dart
    import '../../phonology/domain/word_generator.dart';
    import '../../phonology/data/phonotactic_providers.dart';

    /// Provides a function to validate any word against current phonotactic constraints.
    ///
    /// Usage: ref.watch(phonotacticValidatorProvider)(word: 'some IPA')
    final phonotacticValidatorProvider = Provider<ValidationResult Function({required String word})>((ref) {
      final constraints = ref.watch(parsedConstraintsProvider).asData?.value ?? [];
      final inventory = ref.watch(phonemeInventoryProvider).asData?.value ?? [];
      final generator = WordGenerator();

      return ({required String word}) => generator.validateWord(
        word: word,
        constraints: constraints,
        inventory: inventory,
      );
    });
    ```
